<script>
(function () {
  // Trigger the imported plugin's $output once to build window.weld.skybridge (not auto-invoked on import).
  try { if (typeof root !== 'undefined' && typeof root.weldSkybridge === 'function') root.weldSkybridge(); } catch (e) {}
  var sb = (window.weld && window.weld.skybridge) || null;
  function $(id){ return document.getElementById(id); }
  function setStatus(t, cls){ var e=$('status'); if(e){ e.textContent=t; e.className='status '+(cls||''); } }

  var lastTs = 0, ageTimer = null;
  function render(ch, t){
    if (!ch) return;
    if (ch.hr != null){ var b=$('bpm'); b.textContent=ch.hr; b.classList.add('beat'); setTimeout(function(){ b.classList.remove('beat'); }, 140); }
    $('hrv').textContent = (ch.hrv != null) ? ('HRV ' + ch.hrv + ' ms') : '';
    lastTs = t || Date.now();
    // expose for the generator author to use however they like:
    try { window.__hr = { hr: ch.hr, hrv: ch.hrv, t: lastTs }; } catch (e) {}
  }
  function tickAge(){ if(!lastTs){ $('age').textContent=''; return; } var s=Math.round((Date.now()-lastTs)/1000); $('age').textContent = s<3 ? 'live' : (s+'s ago'); }

  if (!sb){ setStatus('Skybridge plugin not loaded — is {import:weld-skybridge-plugin} in this generator?', 'err'); return; }

  function begin(){
    if (!sb.connected){ setStatus('no Rook companion linked — install/enable the Rook AI extension, then reload', 'off'); return; }
    if (!sb.has('soma')){ setStatus('companion linked but no `soma` capability — update the Rook extension', 'off'); return; }
    setStatus('linked — waiting for a heart-rate sensor…', 'off');
    // 1) pull the latest snapshot once (fills in immediately if a sensor is already streaming)
    sb.request('soma', { op: 'latest' }).then(function (r){
      if (r && r.ok && r.value && r.value.ch && r.value.ch.hr != null){ render(r.value.ch, r.value.t); setStatus('live', 'ok'); }
    });
    // 2) subscribe for live pushes on the soma channel
    if (sb.has('bus')) {
      sb.bus.subscribe('soma:sensors', function (m){ if (m && m.ch){ render(m.ch, m.t); setStatus('live', 'ok'); } });
    }
    // gentle poll as a fallback in case bus consent is declined
    setInterval(function(){ sb.request('soma', { op:'latest' }).then(function(r){ if (r && r.ok && r.value && r.value.ch && r.value.ch.hr != null){ render(r.value.ch, r.value.t); setStatus('live', 'ok'); } }); }, 3000);
    ageTimer = setInterval(tickAge, 1000);
  }

  if (sb.connected) begin();
  else { setStatus('looking for Rook…', 'off'); sb.onConnect(begin); setTimeout(function(){ if (!sb.connected) setStatus('no Rook companion linked — install/enable the Rook AI extension, then reload', 'off'); }, 4000); }
})();
</script>
