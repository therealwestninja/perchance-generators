resetFuraiGalleryFullscreenZoom(overlay) =>
  const img = overlay?.querySelector("[data-furai-gallery-fullscreen-image]")
  if(!img) return

  img.classList.remove("is-secondary-zoomed")
  img.style.setProperty("--furai-gallery-zoom-x", "50%")
  img.style.setProperty("--furai-gallery-zoom-y", "50%")
  img.setAttribute("aria-pressed", "false")
  img.title = "Click to zoom to 200%"

closeFuraiGalleryFullscreen() =>
  const overlay = window._furaiGalleryFullscreenOverlay
  if(!overlay) return

  resetFuraiGalleryFullscreenZoom(overlay)
  overlay.style.display = "none"

showImageFullscreen(imageUrl) =>
  const cleanUrl = String(imageUrl || "").trim()
  if(!cleanUrl) return

  ensureFuraiGalleryFullscreenStyle()

  let overlay = window._furaiGalleryFullscreenOverlay

  if(!overlay) {
    overlay = document.createElement("div")
    overlay.className = "furai-gallery-fullscreen"
    overlay.innerHTML = `
      <button type="button" class="furai-gallery-fullscreen-close" data-furai-gallery-fullscreen-close aria-label="Close">×</button>
      <img class="furai-gallery-fullscreen-image"
        data-furai-gallery-fullscreen-image
        draggable="false"
        role="button"
        tabindex="0"
        aria-pressed="false"
        aria-label="Image preview. Activate to zoom to 200 percent."
        title="Click to zoom to 200%">
    `

    const img = overlay.querySelector("[data-furai-gallery-fullscreen-image]")
    const close = overlay.querySelector("[data-furai-gallery-fullscreen-close]")

    const toggleZoom = event => {
      event.preventDefault()
      event.stopPropagation()

      if(img.classList.contains("is-secondary-zoomed")) {
        resetFuraiGalleryFullscreenZoom(overlay)
        return
      }

      const rect = img.getBoundingClientRect()
      if(!rect.width || !rect.height) return

      const clientX = Number.isFinite(event.clientX) ? event.clientX : rect.left + rect.width / 2
      const clientY = Number.isFinite(event.clientY) ? event.clientY : rect.top + rect.height / 2
      const x = Math.max(0, Math.min(100, ((clientX - rect.left) / rect.width) * 100))
      const y = Math.max(0, Math.min(100, ((clientY - rect.top) / rect.height) * 100))

      img.style.setProperty("--furai-gallery-zoom-x", x + "%")
      img.style.setProperty("--furai-gallery-zoom-y", y + "%")
      img.classList.add("is-secondary-zoomed")
      img.setAttribute("aria-pressed", "true")
      img.title = "Click to return to fit"
    }

    img.addEventListener("click", toggleZoom)
    img.addEventListener("keydown", event => {
      if(event.key !== "Enter" && event.key !== " ") return
      toggleZoom(event)
    })

    overlay.addEventListener("click", event => {
      if(event.target === overlay) closeFuraiGalleryFullscreen()
    })

    close.addEventListener("click", closeFuraiGalleryFullscreen)

    document.addEventListener("keydown", event => {
      if(event.key === "Escape" && overlay.style.display === "flex") {
        closeFuraiGalleryFullscreen()
      }
    })

    document.body.appendChild(overlay)
    window._furaiGalleryFullscreenOverlay = overlay
  }

  const img = overlay.querySelector("[data-furai-gallery-fullscreen-image]")
  resetFuraiGalleryFullscreenZoom(overlay)
  img.src = cleanUrl
  overlay.style.display = "flex"

furaiGalleryCommentHash(text) =>
  text = String(text || "")
  let hash = 0
  for(let i = 0; i < text.length; i++) hash = ((hash << 5) - hash + text.charCodeAt(i)) | 0
  return Math.abs(hash).toString(36)

openFuraiGalleryImageComments(data) =>
  if(!data || !data.imageUrl) return

  if(String(data.prompt || "").includes("U1F507")) {
    showFuraiNoCommentsPopup()
    return
  }

  const galleryIsAdult = data.isNsfw === true || data.isNsfw === "true" || String(data.prompt || "").includes("U1F51E")

    if(!hasAdultPass) {
      if(typeof window.requireFuraiAdultAccess === "function") {
        window.requireFuraiAdultAccess("gallery", () => openFuraiGalleryImageComments(data))
      } else {
        console.error("FurAI adult Waiting Room is unavailable.")

  if (typeof emoji !== "undefined" && emoji && emoji.customEmojis) {
    galleryCustomEmojiUrl = emoji.customEmojis
  } else if (typeof FURAI_CUSTOM_EMOJI_URL === "string" && FURAI_CUSTOM_EMOJI_URL.trim()) {
    galleryCustomEmojiUrl = FURAI_CUSTOM_EMOJI_URL
  }

  let inheritedForumConfig = {}
  if (typeof baseForumConfig === "object" && baseForumConfig) {
    inheritedForumConfig = baseForumConfig
  }

  const fallbackConfig = {
    reverseCommentOrder: true,
    loadFonts: "Open Sans, Inter",
    containerStyle: "width:100%;height:100%;",
    inputAreaStyle: "height:68px;",
