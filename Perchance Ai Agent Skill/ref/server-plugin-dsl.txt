serverPluginCodeHash(text) =>
  let h1 = 0xdeadbeef;
  let h2 = 0x41c6ce57;
  for(let i = 0; i < text.length; i++) {
    const ch = text.charCodeAt(i);
    h1 = Math.imul(h1 ^ ch, 2654435761);
    h2 = Math.imul(h2 ^ ch, 1597334677);
  }
  h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);
  h2 = Math.imul(h2 ^ (h2 >>> 16), 2246822507) ^ Math.imul(h1 ^ (h1 >>> 13), 3266489909);
  return (4294967296 * (2097151 & h2) + (h1 >>> 0)).toString(36);

  class EmulatedServerSocket extends EventTarget {
    static CONNECTING = 0;
    static OPEN = 1;
    static CLOSING = 2;
    static CLOSED = 3;
    CONNECTING = 0;
    OPEN = 1;
    CLOSING = 2;
    CLOSED = 3;
    readyState = 0;
    bufferedAmount = 0;
    binaryType = "blob";
    protocol = "";
    extensions = "";
    onopen = null;

    onerror = null;
    onclose = null;

    constructor(url, generatorName, source, sourceHash) {
      super();
      this.url = url;
      for(const type of ["open", "message", "error", "close"]) {
        this.addEventListener(type, event => {
          const handler = this[`on${type}`];
          if(typeof handler === "function") handler.call(this, event);
        });
      }
      const options = globalThis.__perchanceServerPluginEmulatorOptions;
      import("https://server-plugin.perchance.org/emulator.js?v=2026-07-21-1").then(module => {
        if(this.readyState >= EmulatedServerSocket.CLOSING) return;
        module.connectEmulatedSocket({
          generatorName,
          source,
          sourceHash,
          options,
          peer: {
            attachTransport: transport => { this.#transport = transport; },
            open: () => this.#open(),
            message: (kind, data) => this.#message(kind, data),
            sendAck: bytes => { this.bufferedAmount = Math.max(0, this.bufferedAmount - bytes); },
            close: (code, reason, wasClean) => this.#closed(code, reason, wasClean),
            fail: message => this.#fail(message),
          },
        });
      }).catch(error => this.#fail(String(error?.stack || error)));
    }

    #open() {
      if(this.readyState !== EmulatedServerSocket.CONNECTING) return;
      this.readyState = EmulatedServerSocket.OPEN;
      this.dispatchEvent(new Event("open"));
    }
    #message(kind, data) {
      if(this.readyState === EmulatedServerSocket.CLOSED) return;
      let value = data;
      if(kind === 2) {
        const bytes = data instanceof Uint8Array ? data : new Uint8Array(data);
        const buffer = bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength);
        value = this.binaryType === "arraybuffer" ? buffer : new Blob([buffer]);
      }
      this.dispatchEvent(new MessageEvent("message", {data:value, origin:"", lastEventId:""}));
    }
    #fail(message) {
      if(this.readyState === EmulatedServerSocket.CLOSED) return;
      this.dispatchEvent(new ErrorEvent("error", {message}));
      this.#closed(1011, "local server emulator failed", false);
    }
    #closed(code, reason, wasClean) {
      if(this.readyState === EmulatedServerSocket.CLOSED) return;
      this.readyState = EmulatedServerSocket.CLOSED;
      this.dispatchEvent(new CloseEvent("close", {code, reason, wasClean}));
    }
    send(data) {
      if(this.readyState === EmulatedServerSocket.CONNECTING) throw new DOMException("WebSocket is still in CONNECTING state", "InvalidStateError");
      if(this.readyState !== EmulatedServerSocket.OPEN) return;
      let kind, payload, length;
      if(typeof data === "string") {
        kind = 1; payload = data; length = encoder.encode(data).byteLength;
      } else if(data instanceof ArrayBuffer) {
        kind = 2; payload = new Uint8Array(data.slice(0)); length = payload.byteLength;
      } else if(ArrayBuffer.isView(data)) {
        kind = 2; payload = new Uint8Array(data.buffer.slice(data.byteOffset, data.byteOffset + data.byteLength)); length = payload.byteLength;
      } else if(typeof Blob !== "undefined" && data instanceof Blob) {
        length = data.size;
        this.bufferedAmount += length;
        this.#sendChain = this.#sendChain.then(async () => {
          const bytes = new Uint8Array(await data.arrayBuffer());
          this.#transport.send(2, bytes);
        });
        return;
      } else {
        throw new TypeError("WebSocket.send expects a string, Blob, ArrayBuffer, or ArrayBuffer view");
      }
      this.bufferedAmount += length;
      this.#sendChain = this.#sendChain.then(() => this.#transport.send(kind, payload));
    }
    close(code=1000, reason="") {
      if(code !== 1000 && !(code >= 3000 && code <= 4999)) throw new DOMException("Invalid WebSocket close code", "InvalidAccessError");
      reason = String(reason);
      if(encoder.encode(reason).byteLength > 123) throw new DOMException("WebSocket close reason exceeds 123 UTF-8 bytes", "SyntaxError");
      if(this.readyState >= EmulatedServerSocket.CLOSING) return;
      if(this.readyState === EmulatedServerSocket.CONNECTING) {
        this.readyState = EmulatedServerSocket.CLOSING;
        this.#transport?.abort();
        queueMicrotask(() => {
          if(this.readyState === EmulatedServerSocket.CLOSED) return;
          this.dispatchEvent(new ErrorEvent("error", {message:"WebSocket was closed before the connection was established"}));
          this.#closed(1006, "", false);
        });
        return;
      }
      this.readyState = EmulatedServerSocket.CLOSING;
      this.#sendChain = this.#sendChain.then(() => this.#transport.close(code, reason));
    }
  }

  class ProductionServerSocket extends EventTarget {
    static CONNECTING = 0;
    static OPEN = 1;
    static CLOSING = 2;
    static CLOSED = 3;
    CONNECTING = 0;
    OPEN = 1;
    CLOSING = 2;
    CLOSED = 3;
    readyState = 0;
    binaryType = "blob";

    extensions = "";

    onmessage = null;
    onerror = null;
    onclose = null;
    #ws = null;

    constructor(url, tokenPromise) {
      super();
      this.url = url;
      for(const type of ["open", "message", "error", "close"]) {
        this.addEventListener(type, event => {
          const handler = this[`on${type}`];
          if(typeof handler === "function") handler.call(this, event);
        });
      }
      tokenPromise.then(token => {
        if(this.readyState >= ProductionServerSocket.CLOSING) return;
        let target = url;
        if(token) target += `&t=${encodeURIComponent(token)}`;
        if(embedEnforceForced()) target += "&tenforce=1";

        try { ws = new WebSocket(target); }
        catch(error) { return this.#fail(String(error?.message || error)); }
        ws.binaryType = "arraybuffer";
        this.#ws = ws;
        ws.addEventListener("open", () => {
          if(this.readyState !== ProductionServerSocket.CONNECTING) return;
          this.protocol = ws.protocol;
          this.extensions = ws.extensions;
          this.readyState = ProductionServerSocket.OPEN;
          this.dispatchEvent(new Event("open"));
        });
        ws.addEventListener("message", event => {
          if(this.readyState === ProductionServerSocket.CLOSED) return;
          let value = event.data;
          if(value instanceof ArrayBuffer && this.binaryType !== "arraybuffer") value = new Blob([value]);
          this.dispatchEvent(new MessageEvent("message", {data:value, origin:event.origin, lastEventId:""}));
        });
        ws.addEventListener("error", () => {
          if(this.readyState === ProductionServerSocket.CLOSED) return;
          this.dispatchEvent(new Event("error"));
        });
        ws.addEventListener("close", event => {
          if(event.code === EMBED_BLOCKED_CLOSE_CODE) showEmbedBlockedNotice();
          this.#closed(event.code, event.reason, event.wasClean);
        });
      });
    }

    get bufferedAmount() {
      return this.#ws ? this.#ws.bufferedAmount : 0;
    }
    #fail(message) {
      if(this.readyState === ProductionServerSocket.CLOSED) return;
      this.dispatchEvent(new ErrorEvent("error", {message}));
      this.#closed(1006, "", false);
    }
    #closed(code, reason, wasClean) {
      if(this.readyState === ProductionServerSocket.CLOSED) return;
      this.readyState = ProductionServerSocket.CLOSED;
      this.dispatchEvent(new CloseEvent("close", {code, reason, wasClean}));
    }
    send(data) {
      if(this.readyState === ProductionServerSocket.CONNECTING) throw new DOMException("WebSocket is still in CONNECTING state", "InvalidStateError");
      if(this.readyState !== ProductionServerSocket.OPEN) return;
      this.#ws.send(data);
    }
    close(code=1000, reason="") {
      if(code !== 1000 && !(code >= 3000 && code <= 4999)) throw new DOMException("Invalid WebSocket close code", "InvalidAccessError");
      reason = String(reason);
      if(encoder.encode(reason).byteLength > 123) throw new DOMException("WebSocket close reason exceeds 123 UTF-8 bytes", "SyntaxError");
      if(this.readyState >= ProductionServerSocket.CLOSING) return;
      if(this.readyState === ProductionServerSocket.CONNECTING) {
        this.readyState = ProductionServerSocket.CLOSING;
        // Abort a native socket that is still connecting; if none exists yet,
        // the token continuation sees CLOSING and never creates one.
        try { this.#ws?.close(); } catch {}
        queueMicrotask(() => {
          if(this.readyState === ProductionServerSocket.CLOSED) return;
          this.dispatchEvent(new ErrorEvent("error", {message:"WebSocket was closed before the connection was established"}));
          this.#closed(1006, "", false);
        });
        return;
      }
      this.readyState = ProductionServerSocket.CLOSING;
      this.#ws.close(code, reason);
    }
  }

  function rpcError(message, name="ServerSocketRpcError") {
    const error = new Error(message);
    error.name = name;
    return error;
  }
  function responseHeader(data) {
    if(typeof data !== "string" || !data.startsWith(RPC_RESPONSE_PREFIX)) return null;
    const rest = data.slice(RPC_RESPONSE_PREFIX.length);
    const split = rest.indexOf(":");
    if(split < 1) return null;
    const idText = rest.slice(0, split);
    const kind = rest.slice(split + 1);
    if(!/^[0-9]+$/.test(idText) || !["t", "b", "e"].includes(kind)) return null;
    const id = Number(idText);
    return Number.isSafeInteger(id) && id > 0 && id <= 0xffffffff ? {id, kind} : null;
  }
  function requestFrame(id, method, data) {
    const methodBytes = encoder.encode(method);
    if(methodBytes.byteLength === 0 || methodBytes.byteLength > MAX_RPC_METHOD_BYTES || method.startsWith("@")) {
      throw new TypeError(`invalid RPC method; use 1-${MAX_RPC_METHOD_BYTES} UTF-8 bytes and do not start with '@'`);
    }
    let kind;
    let payload;
    if(typeof data === "string") {
      kind = 1;
      payload = encoder.encode(data);
    } else if(data instanceof ArrayBuffer) {
      kind = 2;
      payload = new Uint8Array(data);
    } else if(ArrayBuffer.isView(data)) {
      kind = 2;
      payload = new Uint8Array(data.buffer, data.byteOffset, data.byteLength);
    } else {
      throw new TypeError("socket.rpc methods expect a string, ArrayBuffer, or ArrayBuffer view");
    }
    if(payload.byteLength + methodBytes.byteLength > MAX_RPC_PAYLOAD_BYTES) throw new TypeError("RPC payload is too large");
    const frame = new Uint8Array(RPC_HEADER_BYTES + methodBytes.byteLength + payload.byteLength);
    frame.set(RPC_REQUEST_MAGIC);
    frame[8] = kind;
    const view = new DataView(frame.buffer);
    view.setUint32(9, id);
    view.setUint16(13, methodBytes.byteLength);
    frame.set(methodBytes, RPC_HEADER_BYTES);
    frame.set(payload, RPC_HEADER_BYTES + methodBytes.byteLength);
    return frame;
  }
  function binaryResult(data) {
    if(data instanceof ArrayBuffer) return Promise.resolve(data);
    if(ArrayBuffer.isView(data)) return Promise.resolve(data.buffer.slice(data.byteOffset, data.byteOffset + data.byteLength));
    if(typeof Blob !== "undefined" && data instanceof Blob) return data.arrayBuffer();
    return Promise.reject(rpcError("invalid binary RPC response", "ServerSocketRpcProtocolError"));
  }

  function settle(id, error, value) {
    const call = pending.get(id);
    if(!call) return;
    pending.delete(id);
    if(error) call.reject(error); else call.resolve(value);
  }
  function rejectAll(message) {
    const error = rpcError(message, "ServerSocketRpcDisconnectedError");
    for(const id of [...pending.keys()]) settle(id, error);
    expectedResponse = null;
  }

  }

      expectedResponse = null;
      const data = event.data;
      receiveChain = receiveChain.then(async () => {
        if(response.kind === "t" || response.kind === "e") {
          if(typeof data !== "string") return protocolFailure(response.id, "invalid text RPC response");
          if(response.kind === "e") settle(response.id, rpcError(data || "server RPC failed", "ServerSocketRpcRemoteError"));
          else settle(response.id, null, data);
          return;
        }
        try { settle(response.id, null, await binaryResult(data)); }
        catch(error) { protocolFailure(response.id, error.message); }
      });
      return;
    }
    const header = responseHeader(event.data);
    if(header !== null) {
      event.stopImmediatePropagation();
      expectedResponse = header;
    }
  }, {capture:true});

  socket.addEventListener("open", () => {
    didOpen = true;
    openedResolve(socket);
  }, {once:true});
  socket.addEventListener("close", event => {
    if(!didOpen) openedReject(rpcError(`WebSocket closed before opening (code ${event.code})`, "ServerSocketOpenError"));
    rejectAll(`WebSocket closed (code ${event.code})`);
    closedResolve(event);
  }, {once:true});

  function callRpc(method, args) {
    if(args.length !== 1) throw new TypeError(`socket.rpc.${method} expects exactly one argument`);
    if(socket.readyState !== WebSocket.OPEN) throw new DOMException("WebSocket is not open", "InvalidStateError");
    if(pending.size >= 0xffffffff) throw rpcError("all RPC request IDs are in use", "ServerSocketRpcQueueFullError");
    while(pending.has(nextId)) nextId = nextId === 0xffffffff ? 1 : nextId + 1;
    const id = nextId;
    nextId = nextId === 0xffffffff ? 1 : nextId + 1;
    const frame = requestFrame(id, method, args[0]);
    const promise = new Promise((resolve, reject) => pending.set(id, {resolve, reject}));
    try { sendRaw(frame); }
    catch(error) { settle(id, error); }
    return promise;
  }

  const methods = new Map();
  const rpc = new Proxy(Object.create(null), {
    get(_target, property) {
      if(property === "then" || property === "toJSON") return undefined;
      if(property === Symbol.toStringTag) return "ServerSocketRpc";
      if(typeof property !== "string") return undefined;
      let method = methods.get(property);
      if(method === undefined) {
        method = (...args) => callRpc(property, args);
        methods.set(property, method);
      }
      return method;
    },
  });

  Object.defineProperties(socket, {
    opened: {value:opened, enumerable:false},
    closed: {value:closed, enumerable:false},
    rpc: {value:rpc, enumerable:false},
  });
  return socket;
