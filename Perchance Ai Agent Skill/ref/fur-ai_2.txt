<script id="furai-boot-protection">
  (() => {
    const getLoader = () => document.getElementById('globalLoader');

    const complete = (delay = 0) => {
      setTimeout(() => {
        const loader = getLoader();
        if (!loader || loader.dataset.hiding === '1') return;
        loader.dataset.hiding = '1';
        loader.style.pointerEvents = 'none';
        loader.style.opacity = '0';
        setTimeout(() => loader.remove(), 110);
      }, Math.max(0, Number(delay) || 0));
    };

    const fail = error => {
      console.error('FurAI boot failed:', error);
      const text = getLoader()?.querySelector('.loading-text');
      if (text) text.textContent = 'Loading...';
    };

    window.FuraiBoot = Object.freeze({ complete, fail });

    window.addEventListener('error', event => {
      const target = event?.target;
      if (!(target instanceof HTMLElement)) return;
      const url = String(target.getAttribute?.('src') || target.getAttribute?.('href') || '');
      if (!/src\/furai\.(?:js|css)(?:[?#]|$)/i.test(url)) return;
      fail(new Error(`Required FurAI file failed to load: ${url}`));
    }, true);
  })();
</script>

<script>
  function initMessageControls() {
    const chatMessagesEl = document.getElementById('chatMessagesEl');
    if (!chatMessagesEl || chatMessagesEl.dataset.controlsReady === '1') return;
    chatMessagesEl.dataset.controlsReady = '1';

    // One shared definition for an unfinished AI-owned bubble.
    // Completed bubbles stay interactive even while another reply is generating.
    window.isFuraiActiveAiMessageElement = window.isFuraiActiveAiMessageElement || function(message) {
      if (!message) return false;
      if (message.classList?.contains('furai-active-ai-message')) return true;
      if (message.id && message.id.startsWith('typing-')) return true;

      const content = message.querySelector?.('.content');
      return !!content && (
        content.classList.contains('typing') ||
        content.classList.contains('furai-message-thinking')
      );
    };

    const isLiveMessageBubble = (message) =>
      !message || window.isFuraiActiveAiMessageElement(message);

    // =====================================================================
    // FORGIVING MESSAGE-CONTROL HOVER CONTRACT — DO NOT REMOVE OR SIMPLIFY
    // =====================================================================
    // The toolbar sits below the visible bubble. A real pointer briefly leaves
    // the bubble while crossing that gap, so immediate CSS-only mouseleave logic
    // makes the toolbar vanish before the user can reach it.
    //
    // Keep one delegated pointerover/pointerout pair on #chatMessagesEl, retain
    // the hover state for a short grace period, and cancel dismissal as soon as
    // the pointer reaches the same message or its toolbar. Do NOT replace this
    // with per-message listeners, immediate mouseleave hiding, repeated timers,
    // or geometry checks. Click selection remains the persistent fallback.
    const messageControlLeaveTimers = new WeakMap();
    const MESSAGE_CONTROL_LEAVE_DELAY_MS = 180;

    const keepMessageControlsHovered = (message) => {
      if (!message) return;

      const pendingTimer = messageControlLeaveTimers.get(message);
      if (pendingTimer) {
        clearTimeout(pendingTimer);
        messageControlLeaveTimers.delete(message);
      }

      message.classList.add('furai-controls-hovered');
    };

    const scheduleMessageControlsLeave = (message) => {
      if (!message) return;

      const pendingTimer = messageControlLeaveTimers.get(message);
      if (pendingTimer) clearTimeout(pendingTimer);

      const timer = setTimeout(() => {
        message.classList.remove('furai-controls-hovered');
        messageControlLeaveTimers.delete(message);
      }, MESSAGE_CONTROL_LEAVE_DELAY_MS);

      messageControlLeaveTimers.set(message, timer);
    };

    chatMessagesEl.addEventListener('pointerover', (event) => {
      if (event.pointerType === 'touch' || document.body.classList.contains('vn-active')) return;

      const message = event.target.closest?.('.message');
      if (
        !message ||
        !chatMessagesEl.contains(message) ||
        message.id === 'personality-selector-message' ||
        isLiveMessageBubble(message)
      ) return;

      keepMessageControlsHovered(message);
    });

    chatMessagesEl.addEventListener('pointerout', (event) => {
      if (event.pointerType === 'touch' || document.body.classList.contains('vn-active')) return;

      const message = event.target.closest?.('.message');
      if (!message || !chatMessagesEl.contains(message)) return;

      const nextTarget = event.relatedTarget;
      if (nextTarget && message.contains(nextTarget)) return;

      scheduleMessageControlsLeave(message);
    });
    
    // --- 1. Master Event Listener (message selection + control buttons) ---
    chatMessagesEl.addEventListener('click', async (e) => {
        const clickedMessage = e.target.closest('.message');
        const clickedInteractive = e.target.closest(
            '.message-controls, button, a, input, textarea, select, [contenteditable="true"], .image-grid-container'
        );

        // =====================================================================
        // CONSOLIDATED MESSAGE-SELECTION CONTRACT — KEEP CR CONTROLS STABLE
        // =====================================================================
        // Clicking any finished bubble selects it. Selection, highlight, and its
        // control bar must remain visible until the user clicks away or selects a
        // different bubble. Do not make control visibility depend only on hover:
        // short and empty bubbles leave no safe pointer path to a detached toolbar.
        // CR selection is applied directly even if VN target resolution later fails.
        if (
            !clickedMessage &&
            !clickedInteractive &&
            !document.body.classList.contains('vn-active')
        ) {
            if (typeof resetVnSceneTarget === 'function') {
                resetVnSceneTarget();
            } else if (typeof syncFuraiCrVnSelectedMessage === 'function') {
                syncFuraiCrVnSelectedMessage(null);
            }

            window.furaiCrSelectionPendingForVn = '';
        }

        if (
            clickedMessage &&
            !clickedInteractive &&
            clickedMessage.id !== 'personality-selector-message' &&
            !isLiveMessageBubble(clickedMessage)
        ) {
            if (typeof syncFuraiCrVnSelectedMessage === 'function') {
                syncFuraiCrVnSelectedMessage(clickedMessage);
            }

            const vnTargetSet = typeof setVnSceneTargetFromElement === 'function' &&
                setVnSceneTargetFromElement(clickedMessage, {
                    autoScroll: false,
                    userLocked: true
                });

            if (vnTargetSet) {
                window.furaiCrSelectionPendingForVn = currentChatId || '';
            }
        }

        // Find the closest button if the user clicked an icon inside it.
        const btn = e.target.closest('.control-btn');
        if (!btn) return;

        let message = btn.closest('.message');
        if (!message) return;

        // Live typing bubbles are not DB messages yet.
        // Controls should only work after completeGeneration() saves the message.
        if (isLiveMessageBubble(message)) return;

        // A. Copy Logic
        if (btn.classList.contains('copy-btn')) {
            const textToCopy = message.querySelector('.content')?.innerText || '';
            if (!textToCopy) return;
            
            // Helper to update icon
            const updateIcon = (success) => {
                if(success) {
                    btn.innerHTML = '<i class="fa-solid fa-check fa-sm"></i>';
                    btn.style.color = "var(--primary-color)";
                    setTimeout(() => {
                        btn.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">content_copy</span>';
                        btn.style.color = "";
                    }, 1000);
                }
            };

            try {
                await navigator.clipboard.writeText(textToCopy);
                updateIcon(true);
            } catch (err) {
                console.warn('Clipboard API blocked, using fallback.');
                const textArea = document.createElement("textarea");
                textArea.value = textToCopy;
                textArea.style.position = "absolute";
                textArea.style.left = "-9999px";
                document.body.appendChild(textArea);
                textArea.select();
                document.execCommand("copy");
                document.body.removeChild(textArea);
                updateIcon(true);
            }
        }
        // B. Edit Logic
        else if (btn.classList.contains('edit-btn')) {
            if(window.editMessage) window.editMessage(message);
        }
        // C. Delete Logic
        else if (btn.classList.contains('delete-btn')) {
            if(window.deleteMessage) window.deleteMessage(message);
        }
        // D. Regenerate Logic
        else if (btn.classList.contains('regen-btn')) {
            await retryFuraiMessageFromElement(message);
        }
        // E. Generate Image Logic
        else if (btn.classList.contains('gen-img-btn')) {
            const promptText = message.querySelector('.content')?.innerText || '';

            if (window.generateImagesForMessage) {
                window.generateImagesForMessage(message, promptText);
            }
        }
        // F. Save to Concept Board
        else if (btn.classList.contains('concept-save-btn')) {
            if (window.saveChatBubbleToConceptBoard) {
                window.saveChatBubbleToConceptBoard(btn, message);
            }
        }
        // G. Pin to Key Facts
        else if (btn.classList.contains('pin-fact-btn')) {
            if (window.togglePinnedFactForMessage) {
                window.togglePinnedFactForMessage(message);
            }
        }
    });

    // --- 2. Lightweight Element Creator (No listeners attached here) ---
    window.placeMessageControls = function(message, controlsContainer) {
      if (!message) return;

      const controls = controlsContainer || message.querySelector('.message-controls');
      if (!controls) return;

      // One anchor for every bubble shape: the content wrapper owns the toolbar.
      // Never append ordinary controls to the outer .message row again. That row
      // also contains avatars, so one-line and empty bubbles would place controls
      // too low and make the toolbar jump when bubble geometry changed.
      const wrapper = message.querySelector('.message-content-wrapper');
      const grid = wrapper ? wrapper.querySelector('.image-grid-container') : null;

      if (wrapper) {
        if (grid) {
          wrapper.insertBefore(controls, grid);
        } else {
          wrapper.appendChild(controls);
        }
        return;
      }

      // Defensive fallback for malformed/legacy rendered nodes only.
      message.appendChild(controls);
    };
    
    window.addMessageControls = function(message, retryRecordOverride = null) {
      if (
        !message ||
        message.querySelector('.message-controls') ||
        message.id === 'personality-selector-message' ||
        isLiveMessageBubble(message)
      ) return;

      // Bulk chat rendering already has the exact stored message in memory.
      // Reuse it instead of reopening/decompressing the whole canonical chat once
      // per bubble just to decide which controls should be shown. DOM-only legacy
      // callers still use the canonical lookup as a safe fallback.
      const retryRecord = retryRecordOverride || (
          typeof furaiGetMessageAtDomElement === 'function'
              ? furaiGetMessageAtDomElement(message)
              : null
      );
      const fallbackRole = getFuraiMessageFrameRole(message);
      const retryMessage = retryRecord?.msg || {
          role: fallbackRole,
          userUid: message.dataset.userUid || '',
          botUid: message.dataset.botUid || '',
          botIndex: message.dataset.botIndex,
          speakerName: message.dataset.speakerName || ''
      };

      // A true nameless system notice remains immutable. Human turns, AI turns,
      // Silent Notes, Cutscenes, and anonymous user broadcasts all receive Retry.
      if (isFuraiTrueNamelessSystemMessage(retryMessage, message)) return;

      const controlsContainer = document.createElement('div');
      controlsContainer.className = 'message-controls';

      // --- Gen Image Button Logic ---
      // UNLOCKED: Generate image button is now available for all messages in all modes
      const genButton = document.createElement('button');
      genButton.className = 'control-btn gen-img-btn';
      genButton.title = 'Generate Images';
      genButton.innerHTML = '<span class="material-symbols-outlined" style="font-size:18px; line-height:1; vertical-align:middle;">mms</span>';
      controlsContainer.appendChild(genButton);

      // Standard Buttons
      const copyButton = document.createElement('button');
      copyButton.className = 'control-btn copy-btn';
      copyButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">content_copy</span>';
      copyButton.title = 'Copy';

      const editButton = document.createElement('button');
      editButton.className = 'control-btn edit-btn';
      editButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">stylus</span>';
      editButton.title = 'Edit';

      const deleteButton = document.createElement('button');
      deleteButton.className = 'control-btn delete-btn';
      deleteButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">delete</span>';
      deleteButton.title = 'Delete';

      const conceptSaveButton = document.createElement('button');
      conceptSaveButton.className = 'control-btn concept-save-btn';
      conceptSaveButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">note_stack_add</span>';
      conceptSaveButton.title = 'Save to Concept Board';

      const pinButton = document.createElement('button');
      pinButton.className = 'control-btn pin-fact-btn';
      pinButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">push_pin</span>';
      pinButton.title = 'Pin to Key Facts';

      const retryKind = getFuraiMessageRetryKind(retryMessage, message);
      if (retryKind !== 'none') {
          const regenButton = document.createElement('button');
          regenButton.className = 'control-btn regen-btn';
          regenButton.innerHTML = '<span class="material-symbols-outlined" style="font-size: 18px; vertical-align: middle;">autorenew</span>';
          regenButton.title = getFuraiRetryButtonTitle(retryMessage, message);
          controlsContainer.appendChild(regenButton);
      }

      const variantNav = typeof window.createMessageVariantNav === 'function'
          ? window.createMessageVariantNav(message, retryMessage)
          : null;

      if (variantNav) {
          controlsContainer.appendChild(variantNav);
      }

      controlsContainer.appendChild(copyButton);
      controlsContainer.appendChild(editButton);
      controlsContainer.appendChild(pinButton);
      controlsContainer.appendChild(conceptSaveButton);
      controlsContainer.appendChild(deleteButton);

      window.placeMessageControls(message, controlsContainer);
    };

    // --- 3. Optimized Observer (subtree: false) ---
    const observer = new MutationObserver(mutations => {
      mutations.forEach(mutation => {
        mutation.addedNodes.forEach(node => {
          if (node.nodeType === 1 && node.classList.contains('message')) {
            addMessageControls(node);
          }
        });
      });
    });

    // KEY FIX: subtree: false prevents checking every tiny change inside the message
    observer.observe(chatMessagesEl, { childList: true, subtree: false });

    // Initial load
    chatMessagesEl.querySelectorAll('.message').forEach(addMessageControls);
  }
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initMessageControls, { once: true });
  } else {
    setTimeout(initMessageControls, 0);
  }
</script>
