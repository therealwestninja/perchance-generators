upload(data, opts) =>
  if(!opts) opts = {};
  // if(typeof data !== "string") return alert("error: only text/strings are supported by the upload plugin for now. please request other types on the forum.");
  
  ensureUploadPluginIframe();
  
  let requestId = "id"+(Math.random().toString()+Math.random().toString()).replaceAll(".", "");
  
  let promiseResolver, promiseRejecter;
  let promise = new Promise((resolve, reject) => {
    promiseResolver = resolve;
    promiseRejecter = reject; // unused.
  });
  
  function messageHandler(event) {
    if(event.source !== window.__uploadPluginIframe0435342908.contentWindow) return;
    if(event.origin !== "https://upload.perchance.org") return;

    if(event.data.type === "anonUploadResponse" && event.data.requestId === requestId) {
      let url = event.data.result.url;
      let error = event.data.result.error;
      let message = event.data.result.message;
      let size = event.data.result.size;
      let deletionUrl = event.data.result.deletionUrl;
      
      if(!error) error = null; // just make sure it's null rather than 'undefined', since that's part of the specified behavior
      
      // If it didn't upload successfully, then deletionUrl can just be a dummy URL that returns success:
      if(!deletionUrl) deletionUrl = "https://user.uploads.dev/file/fcf0aa53193434e1de2a3a2b878bb893.txt";

      // Reduce breaking-change-ness of the move to user.uploads.dev for a common pattern of getting the file id by removing the URL's prefix:
      if(typeof url === "string") {
        let originalUrl = url;
        url = new String(url)
        url.replace = function(...args) {
          if(typeof args[0] === "string" && args[0].startsWith("https://user-uploads.perchance.org")) {
            args[0] = args[0].replace("https://user-uploads.perchance.org", "https://user.uploads.dev");
          }
          return originalUrl.replace(...args);
        };
      }
      
      promiseResolver({url, error, size, deletionUrl}); // always 'resolve' (i.e. we don't `reject` on error). easier for people to handle errors without knowing about try/catch stuff - they just check if `error` is null
      window.removeEventListener('message', messageHandler);
      
      let el = document.querySelector(`#${requestId}`);
      if(el) {
        if(error) {
          el.innerHTML = error;
        } else {
          el.innerHTML = `<a href="${url}" target="_blank">${url}</a>`;
        }
      }
      if(error) {
        let div = document.createElement("div");
        let errorMessageHtml = message ? ` (${message})` : "";
        div.innerHTML = `<div style="z-index:1000; position: fixed; bottom: 1rem; width: 100%; pointer-events:none;"><span style=" padding: 0.5rem; background: #ac2c2c; border-radius: 3px; color: white; pointer-events:auto;">upload error: ${error.replaceAll("_", " ")}${errorMessageHtml}</span></div>`;
        div = div.firstElementChild;
        document.body.appendChild(div);
        setTimeout(() => {
          div.remove();
        }, 1000*5);
      }
    }
  }
  window.addEventListener('message', messageHandler);
  
  (async function() {
    while(!window.__uploadPluginEmbedIsReady) {
      console.log("waiting for upload iframe embed to be ready");
      await new Promise(r => setTimeout(r, 100));
    }
    
    function blobToDataUrl(blob) {
      return new Promise((resolve, _) => {
        const reader = new FileReader();
        reader.onloadend = () => resolve(reader.result);
        reader.readAsDataURL(blob);
      });
    }
    
    let dataUrl;
    if(typeof data === "string") dataUrl = `data:text/plain;charset=utf-8,${encodeURIComponent(data)}`;
    if(data instanceof Blob) dataUrl = await blobToDataUrl(data);
    
    window.__uploadPluginIframe0435342908.contentWindow.postMessage({type:"anonUploadRequest", requestId, dataUrl, expires:opts.expires, generatorName:window.generatorName}, 'https://upload.perchance.org');
  })();
  
  promise.toString = function() {
    return `<span id="${requestId}">uploading...</span>`;
  };
  return promise;

  let editable = getEditableObj();
  let nativeBind = Function.prototype.bind;
  function attachEditable(fn) {
    Object.defineProperty(fn, "editable", {value:editable, configurable:true});
    Object.defineProperty(fn, "bind", {
      value:function(...args) {
        return attachEditable(nativeBind.apply(this, args));
      },
      configurable:true,
    });
    return fn;
  }
  return attachEditable(upload);


ensureUploadPluginIframe() =>

    window.__alreadyLoadedUploadPluginStuff0435342908 = true;
    
    window.__uploadPluginIframe0435342908 = document.createElement('iframe'); 

    window.addEventListener('message', (event) => {
      if(event.source !== window.__uploadPluginIframe0435342908.contentWindow) return;
      if(event.origin !== "https://upload.perchance.org") return;
      console.log("upload plugin got message:", event);

        window.__uploadPluginEmbedIsReady = true;

      }
    });

    function initIframeAttributes(iframe) {

      iframe.style.cssText = 'width:1px; height:1px; border:none; top:-100px; left:-100px; position:fixed;';
    }
    initIframeAttributes(window.__uploadPluginIframe0435342908);
    document.body.appendChild(window.__uploadPluginIframe0435342908);

    // ping embed until it is ready (hacky way to wait for it to load)
    (async function() {
      await new Promise(r => setTimeout(r, 500));
      let waitedTime = 0;
      while(!window.__uploadPluginEmbedIsReady) {
        if(waitedTime > 1000*20) { // if we've waited too long (perhaps due to upload server error response), try deleting the iframe and adding it again
          window.__uploadPluginIframe0435342908.remove();
          window.__uploadPluginIframe0435342908 = document.createElement('iframe');
          initIframeAttributes(window.__uploadPluginIframe0435342908);
          document.body.appendChild(window.__uploadPluginIframe0435342908);
        }
        await new Promise(r => setTimeout(r, 500));
        waitedTime += 500;
        window.__uploadPluginIframe0435342908.contentWindow.postMessage({type:"init"}, 'https://upload.perchance.org');
      }
    })();
  }


getEditableObj() =>
  function editableUrl(name) {
    if(typeof name !== "string" || !/^[a-z0-9-]{1,200}$/.test(name)) {
      throw new TypeError("Editable file names must be 1 to 200 lowercase letters, numbers, or hyphens.");
    }
    return `https://editable.uploads.dev/file/${window.generatorName}/${name}`;
  }

  let editable = {};
  editable.get = async function(name) {
    let response = await fetch(editableUrl(name));
    if(response.status === 404) return null;
    if(!response.ok) throw new Error(`Failed to fetch editable file: HTTP ${response.status}`);
    return await response.text();
  };

  editable.set = function(name, data, opts) {
    if(!opts) opts = {};
    if(typeof window.generatorPublicId !== "string" || !/^[0-9a-f]{32}$/.test(window.generatorPublicId)) {
      return Promise.resolve({url:null, size:null, editKey:null, error:"editable_requires_saved_generator"});
    }
    if(typeof name !== "string" || !/^[a-z0-9-]{1,200}$/.test(name)) {
      return Promise.resolve({url:null, size:null, editKey:null, error:"invalid_editable_name"});
    }
    if(typeof data !== "string") {
      return Promise.resolve({url:null, size:null, editKey:null, error:"invalid_data_type"});
    }
    if(new TextEncoder().encode(data).byteLength > 1024*1024*5) {
      return Promise.resolve({url:null, size:null, editKey:null, error:"file_too_big"});
    }
    if(typeof opts !== "object" || Array.isArray(opts)) {
      return Promise.resolve({url:null, size:null, editKey:null, error:"invalid_options"});
    }
    if(opts.editKey !== undefined && typeof opts.editKey !== "string") {
      return Promise.resolve({url:null, size:null, editKey:null, error:"invalid_edit_key"});
    }

    ensureUploadPluginIframe();
    let requestId = "editable"+(Math.random().toString()+Math.random().toString()).replaceAll(".", "");
    let promiseResolver;
    let promise = new Promise(resolve => promiseResolver = resolve);

    function messageHandler(event) {
      if(event.source !== window.__uploadPluginIframe0435342908.contentWindow) return;
      if(event.origin !== "https://upload.perchance.org") return;
      if(event.data.type !== "editableSetResponse" || event.data.requestId !== requestId) return;

      window.removeEventListener("message", messageHandler);
      let result = event.data.result || {};
      promiseResolver({
        url: result.url || null,
        size: typeof result.size === "number" ? result.size : null,
        editKey: result.editKey || null,
        editCount: typeof result.editCount === "number" ? result.editCount : null,
        created: result.created === true,
        unchanged: result.unchanged === true,
        superseded: result.superseded === true,
        error: result.success ? null : (result.status || "unknown_error"),
      });
    }
    window.addEventListener("message", messageHandler);

    (async function() {
      while(!window.__uploadPluginEmbedIsReady) {
        await new Promise(resolve => setTimeout(resolve, 100));
      }
      window.__uploadPluginIframe0435342908.contentWindow.postMessage({
        type: "editableSetRequest",
        requestId,
        generatorName: window.generatorName,
        name,
        data,
        editKey: opts.editKey,
      }, "https://upload.perchance.org");
    })();

    return promise;
  };
  return editable;
